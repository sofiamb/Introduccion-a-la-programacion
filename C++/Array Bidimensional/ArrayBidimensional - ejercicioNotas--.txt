#include <iostream>
#include <string>
 
using namespace std;
 
int main() {
    // Definimos 3 alumnos y 3 notas para cada uno
    const int ALUMNOS = 3;
    const int NOTAS = 3;
    
    string nombres[ALUMNOS] = {"Ana", "Luis", "Marta"};
    
    // Matriz: [fila/alumno][columna/nota]
    float calificaciones[ALUMNOS][NOTAS] = {
        {8.5, 9.0, 7.8}, // Notas de Ana
        {6.0, 7.5, 8.0}, // Notas de Luis
        {9.5, 10.0, 9.2} // Notas de Marta
    };
 
    cout << "--- PROMEDIOS DE CLASE ---" << endl;
 
    for (int i = 0; i < ALUMNOS; i++) {
        float suma = 0;
        
        for (int j = 0; j < NOTAS; j++) {
            suma += calificaciones[i][j]; // Sumamos las notas de la fila i
        }
        
        float promedio = suma / NOTAS;
        cout << "Alumno: " << nombres[i] << " | Promedio: " << promedio << endl;
    }
 
    return 0;
}
