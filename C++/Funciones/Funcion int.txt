#include <iostream>
#include <string>

using namespace std;

int valorIncrementado(int _value) {// int - float - bool
	_value++;
	return _value;
}


int main()
{
	int n = 1;
	int numero = valorIncrementado(n); 

	cout << "numero vale: " << numero<<"\n";


}
